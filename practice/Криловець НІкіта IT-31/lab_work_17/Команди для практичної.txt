
-- =========================================
-- ЗАВДАННЯ 1. НАЇВНИЙ ІМПОРТ
-- =========================================

CREATE TABLE medicines_staging (
    name TEXT,
    manufacturer TEXT,
    form TEXT,
    price REAL,
    stock_quantity INTEGER
);

.mode csv
.import --skip 1 v4_raw_rows.csv medicines_staging

-- Перевіряємо кількість рядків
SELECT COUNT(*) FROM medicines_staging;

-- Переглядаємо перший запис
SELECT * FROM medicines_staging LIMIT 1;

-- Очищаємо проміжну таблицю перед правильним імпортом
DELETE FROM medicines_staging;


-- =========================================
-- ЗАВДАННЯ 2. ПРАВИЛЬНИЙ ІМПОРТ
-- =========================================

.mode csv
.separator ";"

.import --skip 1 v4_raw_rows.csv medicines_staging

-- Перевіряємо кількість записів
SELECT COUNT(*) FROM medicines_staging;

-- Переглядаємо імпортовані дані
SELECT * FROM medicines_staging;


-- =========================================
-- ЗАВДАННЯ 3. ПЕРЕВІРКА ЯКОСТІ ДАНИХ
-- =========================================

-- Перевірка порожніх значень
SELECT
    SUM(name = '') AS name_empty,
    SUM(manufacturer = '') AS manufacturer_empty,
    SUM(form = '') AS form_empty,
    SUM(price = '') AS price_empty,
    SUM(stock_quantity = '') AS stock_empty
FROM medicines_staging;

-- Перевірка типів числових колонок
SELECT
    typeof(price) AS price_type,
    typeof(stock_quantity) AS stock_type,
    COUNT(*) AS amount
FROM medicines_staging
GROUP BY typeof(price), typeof(stock_quantity);

-- Пошук дублікатів
SELECT
    name,
    manufacturer,
    form,
    price,
    stock_quantity,
    COUNT(*) AS amount
FROM medicines_staging
GROUP BY name, manufacturer, form, price, stock_quantity
HAVING COUNT(*) > 1;

-- Кількість унікальних записів
SELECT COUNT(*)
FROM (
    SELECT DISTINCT
        name, manufacturer, form, price, stock_quantity
    FROM medicines_staging
);

-- Перевірка обмежень основної таблиці
.schema medicines


-- =========================================
-- ЗАВДАННЯ 4. ПЕРЕНЕСЕННЯ ДАНИХ
-- =========================================

-- Запам'ятовуємо кількість записів і останній id ДО перенесення
CREATE TEMP TABLE import_info AS
SELECT
    COUNT(*) AS old_count,
    COALESCE(MAX(id), 0) AS old_max_id
FROM medicines;

SELECT * FROM import_info;

-- Переносимо унікальні записи.
-- Порожні значення перетворюємо на NULL.
-- Рядки без назви не переносимо, оскільки name має бути заповнена.

INSERT INTO medicines (
    name,
    manufacturer,
    form,
    price,
    stock_quantity
)
SELECT DISTINCT
    NULLIF(TRIM(name), ''),
    NULLIF(TRIM(manufacturer), ''),
    NULLIF(TRIM(form), ''),
    NULLIF(TRIM(price), ''),
    NULLIF(TRIM(stock_quantity), '')
FROM medicines_staging
WHERE NULLIF(TRIM(name), '') IS NOT NULL;

-- Перевіряємо кількість записів після перенесення
SELECT COUNT(*) FROM medicines;

-- Перевіряємо нові записи
SELECT *
FROM medicines
WHERE id > (SELECT old_max_id FROM import_info);

-- Перевіряємо NULL у нових записах
SELECT *
FROM medicines
WHERE id > (SELECT old_max_id FROM import_info)
AND (
    manufacturer IS NULL
    OR form IS NULL
    OR price IS NULL
    OR stock_quantity IS NULL
);

-- Видаляємо проміжну таблицю
DROP TABLE medicines_staging;


-- =========================================
-- ЗАВДАННЯ 5. ЕКСПОРТ У CSV
-- =========================================

.headers on
.mode csv

.output medicines_export.csv

SELECT
    id,
    name,
    manufacturer,
    form,
    price,
    stock_quantity
FROM medicines
WHERE id > (SELECT old_max_id FROM import_info);

.output stdout

-- Видаляємо тимчасову таблицю з інформацією
DROP TABLE import_info;
