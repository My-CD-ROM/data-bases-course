CREATE TABLE sales_new (
    id INTEGER PRIMARY KEY,
    medicine_id INTEGER NOT NULL,
    customer_id INTEGER,
    sale_date TEXT,
    quantity INTEGER NOT NULL DEFAULT 1,

    FOREIGN KEY (medicine_id) REFERENCES medicines(id) ON DELETE RESTRICT,
    FOREIGN KEY (customer_id) REFERENCES customers(id) ON DELETE SET NULL
);

INSERT INTO sales_new SELECT * FROM sales;
DROP TABLE sales;
ALTER TABLE sales_new RENAME TO sales;

Добавляємо значення до таблиці не вказавши значення та властивість для quantity:
INSERT INTO sales (medicine_id, customer_id, sale_date) 
VALUES (1, 1, '2026-09-23');

Перевіряємо, що quantity автоматично підставило значення 1:
SELECT * FROM sales WHERE id = (SELECT MAX(id) FROM sales);
