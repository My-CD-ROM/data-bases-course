CREATE TABLE IF NOT EXISTS adverse_reactions (
    id INTEGER PRIMARY KEY,
    medicine_id INTEGER NOT NULL,
    reaction_type TEXT NOT NULL,
    severity TEXT NOT NULL
        CHECK (severity IN ('Низька', 'Середня', 'Висока')),
    reaction_date TEXT NOT NULL,
    description TEXT,
    FOREIGN KEY (medicine_id) REFERENCES medicines(id)
);

Завдання 2:
SELECT medicines.id, medicines.name,
    COUNT(adverse_reactions.id) AS reactions_count
FROM medicines
JOIN adverse_reactions
    ON medicines.id = adverse_reactions.medicine_id
GROUP BY medicines.id;


Завдання 3:
SELECT medicines.id, medicines.name,
    COUNT(adverse_reactions.id) AS reactions_count
FROM medicines
JOIN adverse_reactions
    ON medicines.id = adverse_reactions.medicine_id
GROUP BY medicines.id
HAVING COUNT(adverse_reactions.id) > 2;


Завдання 4:
SELECT medicines.id, medicines.name,
    COUNT(adverse_reactions.id) AS serious_reactions
FROM medicines
JOIN adverse_reactions
    ON medicines.id = adverse_reactions.medicine_id
WHERE adverse_reactions.severity = 'Висока'
GROUP BY medicines.id
HAVING COUNT(adverse_reactions.id) > 1;
