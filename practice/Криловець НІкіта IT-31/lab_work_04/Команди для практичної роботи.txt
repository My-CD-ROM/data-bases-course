CREATE TABLE sales (
    id INTEGER PRIMARY KEY,
    medicine_id INTEGER NOT NULL,
    customer_id INTEGER,
    sale_date TEXT,
    quantity INTEGER,

    FOREIGN KEY (medicine_id)
        REFERENCES medicines(id)
        ON DELETE RESTRICT,

    FOREIGN KEY (customer_id)
        REFERENCES customers(id)
        ON DELETE SET NULL
);
